\begin{frame}[fragile]{RegExp implementation: data type}
		\begin{code}
data RegExp = Empty | Epsilon | R [Symbol] | Union RegExp RegExp 
              | Star RegExp | Con RegExp RegExp | Plus RegExp 
  deriving (Show, Eq)
		\end{code}
	Design choices: \pause
	\begin{itemize}
\item not split empty from non-empty expressions	
\item \texttt{Epsilon} versus \texttt{R []}
\item \texttt{R [Symbol]} versus \texttt{R Symbol} 
	\end{itemize}
For convenience:
\begin{code}
regExpUnion :: [RegExp] -> RegExp
regExpUnion = simplify . foldr Union Empty
\end{code}		

\end{frame}
\begin{frame}[fragile]{RegExp but prettier}
	
\begin{code}
pRegExp :: RegExp -> String
pRegExp Empty = ""
pRegExp Epsilon = "R e"
pRegExp (R [])  = "R e" 
pRegExp (R xs) = show xs
pRegExp (Union r1 r2) = "(" ++ pRegExp r1 ++ "|" ++ pRegExp r2 ++ ")"
pRegExp (Star r) = "(" ++  pRegExp r ++ ")*"
pRegExp (Con r1 r2) = pRegExp r1 ++ pRegExp r2
pRegExp (Plus r) = "(" ++ pRegExp r ++ ")+"

ppRegExp :: RegExp -> IO ()
ppRegExp = print . pRegExp
\end{code}	

\texttt{ghci> ppRegExp (Con (Union (R "53") (R [])) (Star (R "19")))}

then becomes 

\texttt{"("53"|R e)("19")*"}
\end{frame}

\begin{frame}[fragile]{RegExp to Generate Words}
	\begin{code}
generateString :: RegExp -> Gen String
generateString = (`generateString'` 5) . simplify
	where generateString' :: RegExp -> Int -> Gen String
	generateString' Empty _ = error "cannot generate from empty language" 
	generateString' Epsilon _ = return ""
	generateString' (R xs) _ = return xs
	generateString' (Union r1 r2) n = oneof [generateString' r1 n, generateString' r2 n]
	generateString' (Con r1 r2) n = do
		w <- generateString' r1 n
		v <- generateString' r2 n
		return (w ++ v)
	generateString' (Star _) 0 = return ""    
	generateString' (Star r) n = oneof [generateString' Epsilon n, generateString' (Con r (Star r)) (n - 1)]
	generateString' (Plus r) n = generateString' (Con r (Star r)) n
	\end{code}
\end{frame}

\begin{frame}[fragile]{And Simplify?}
	\begin{itemize}
		\item Minimising regular expressions is (allegedly) PSPACE-hard
		\item So we did something a bit simpler...
	\end{itemize}
\pause	
\begin{code}
simplify :: RegExp -> RegExp
simplify r | r == simplify' r = r
| otherwise = simplify $ simplify' r
where
simplify' :: RegExp -> RegExp
simplify' Empty = Empty
simplify' Epsilon = Epsilon
simplify' (R xs) = R xs
simplify' (Con Empty Empty) = Empty
simplify' (Con Epsilon Epsilon) = Epsilon
simplify' (Con _ Empty) = Empty
simplify' (Con Empty _) = Empty
simplify' (Con r' Epsilon) = simplify' r'
simplify' (Con Epsilon r') = simplify' r'
simplify' (Con r1 r2) = Con (simplify' r1) (simplify' r2)
simplify' (Union Empty Empty) = Empty
simplify' (Union Empty r') = simplify' r'
simplify' (Union r' Empty) = simplify' r'
simplify' (Union Epsilon Epsilon) = Epsilon
simplify' (Union r1 r2) | r1 /= r2 = Union (simplify' r1) (simplify' r2)
| otherwise = simplify' r1
simplify' (Plus Epsilon) = Epsilon
simplify' (Star Epsilon) = Epsilon
simplify' (Plus Empty) = Empty
simplify' (Star Empty) = Epsilon
simplify' (Star r') = Star (simplify' r')
simplify' (Plus r') = Plus (simplify' r')
\end{code}	
\end{frame}
\begin{frame}[fragile]{RegExp to DFA}
	\begin{itemize}		
		\item For each RegExp $R$ there is a DFA that accepts exactly all words from $R$.\pause
		\item But how? \pause
		\item Recursively!
		\item First we convert regular expressions to $\epsilon$-NFAs
		\item Then we just use the power set construction
	\end{itemize}
\end{frame}
\begin{frame}[fragile]{RegExp to $\epsilon$-NFA: the Base Cases}
	\begin{code}
regExpToENFA :: RegExp -> ENFA Int
regExpToENFA Empty   = ENFA [1] [] (\_ _ -> []) [] 1 []
regExpToENFA Epsilon = ENFA [1] [] (\_ _ -> []) [] 1 [1]
regExpToENFA (R [])  = ENFA [1] [] (\_ _ -> []) [] 1 [1]
regExpToENFA (R xs)  = ENFA [0..length xs] (nub xs) delta2 [] 0 [length xs] 	where
	delta2 symbol state | state >= length xs || state < 0 = []
	| xs !! state == symbol = [state + 1]
	| otherwise = []
	\end{code}
\end{frame}
\begin{frame}[fragile]{RegExp to $\epsilon$-NFA: The Recursion}
	\begin{code}
regExpToENFA (Union r1 r2) = regExpToENFA r1 `unionENFA` makeDisjoint (regExpToENFA r1) (regExpToENFA r2)
regExpToENFA (Star r) = starENFA (regExpToENFA r)
regExpToENFA (Con r1 r2) = regExpToENFA r1 `concatENFA` makeDisjoint (regExpToENFA r1) (regExpToENFA r2)
regExpToENFA (Plus r) = regExpToENFA r `concatENFA` makeDisjoint (regExpToENFA r) (starENFA (regExpToENFA r))
	\end{code}
\end{frame}
\begin{frame}[fragile]{Combining $\epsilon$-NFA-s: Union}
	\begin{code}
unionENFA :: ENFA Int -> ENFA Int -> ENFA Int-- Use only if states are disjoint
unionENFA n1 n2 = ENFA states'' alphabet2 delta'' epT start'' accept where
	states'' = start'' : statesENF n1 ++ statesENF n2
	alphabet2 = alphabetENF n1 `union` alphabetENF n2
	delta'' sym st = deltaENF n1 sym st ++ deltaENF n2 sym st
	epT = epTrans n1 ++ epTrans n2 ++ [(start'', startENF n1), (start'', startENF n2)]
	start'' = maximum (statesENF n2 ++ statesENF n1) + 1 
	accept = acceptstateENF n1 ++ acceptstateENF n2
	\end{code}
\pause	
\begin{itemize}
	\item Concat and Star are handled similarly
	\item Disjoint? 
\end{itemize}

\begin{frame}[fragile]{DFA to RegExp}
	\begin{itemize}
		\item<1> And now for the fun part!
	\end{itemize}
\pause	
	\begin{code}		
dfaToRegExp :: Ord a => DFA a -> RegExp
dfaToRegExp dfa = simplify $ regExpUnion [rijk dfaInt (start dfaInt) f (length $ states dfaInt) | f <- acceptstate dfaInt ]
	where dfaInt = (makeIntDFA . minimizeDFA) dfa
	\end{code}
\pause	
	\begin{itemize}
		\item The idea is as follows: 
		\item First we make the automaton as small as possible.\pause
		\item Then we rename the states a little (to consequetive \texttt{Int [1..length \$ states dfa]}).\pause
		\item Then the magic happens.
	\end{itemize}	
\end{frame}
\begin{frame}[fragile]{Minimising a DFA: Brzozowski's algorithm}
	\begin{itemize}
		\item Reverse the DFA.
		\item Cutting out unreachable states by forward searching from the starting state.
		\item Reverse again.
		\item Cut out unreachable states again.
	\end{itemize}
\begin{code}
reverseDFA :: Ord a => DFA a -> DFA [Int]
reverseDFA d = enfaToDFA' $ ENFA sts alph delt ep st acc where	
e = singleAccept $ cutDFA d
sts = statesENF e
alph = alphabetENF e
delt sym s = [t | t <- statesENF e, s `elem` deltaENF e sym t] 
ep = [(s, t) | (t,s) <- epTrans e ]
st = head $ acceptstateENF e
acc = [startENF e]
\end{code}	
\item Due to the nature of the power set construction and the fact that all states now contribute at least some word, the DFA is now minimal.
\end{frame}
\begin{frame}[fragile]{The Magic}
\begin{code}
rijk :: DFA Int -> Int -> Int -> Int -> RegExp
rijk dfa i j 0 | i == j  = regExpUnion $ Epsilon : labels
| null labels = Empty
| length labels == 1 = head labels
| otherwise = foldr Union (head labels) (tail labels)          
where labels =  [R [x] | x <- alphabet dfa, delta dfa x i == j]
rijk dfa i j k = Union (rijk dfa i j (k-1)) (Con (rijk dfa i k (k-1)) (Con (Star $ rijk dfa k k (k-1)) (rijk dfa k j (k-1))))
\end{code}
\begin{itemize}
	\item \texttt{rijk} stands for $R^k_{ij}$.
	\item $i$ and $j$ are indices of states (hence the renaming).
	\item The $k$ indicates that we only consider states between $i$ and $j$ of index at most $k$.
	\item $R^0_{ij}$ thus covers all paths that do not pass through states. 
	\item $R^{k}_{i j} := R^{k-1}_{ij}+R^{k-1}_{ik}(R^{k-1}_{kk})^*R^{k-1}_{kj}$
\end{itemize}
\end{frame}
\begin{frame}[fragile]{The Magic: continued}
	$R^{k}_{i j} := R^{k-1}_{ij}+R^{k-1}_{ik}(R^{k-1}_{kk})^*R^{k-1}_{kj}$:
	\begin{itemize}
		\item Suppose we have a path from $i$ to $j$ that does not visit states higher than $k$.
		\item $R^{k-1}_{ij}$ covers the option that the path does not visit $k$.
		\item If the path does visit $k$, then:
		\item $R^{k-1}_{ik}$ covers the path from $i$ to $k$.
		\item $(R^{k-1}_{kk})^*$ covers all (already found) paths from $k$ to itself.
		\item $R^{k-1}_{kj}$ covers the path from $j$ to $k$.\pause
		\item Hence $R^{k}_{ij}$ is the regular expression covering all paths between $i$ and $j$ not visiting states with index higher than $k$.
	\end{itemize}
\end{frame}
\begin{frame}[fragile]{The Magic: the End}
recall:
\begin{code}
dfaToRegExp dfa = simplify $ regExpUnion [rijk dfaInt (start dfaInt) f (length $ states dfaInt) | f <- acceptstate dfaInt ]
\end{code}
\begin{itemize}
	\item $i$ is the initial state
	\item $j$ is an accepting state
	\item $k$ is the amount of states
	\item We take the union of $R^k_{ij}$ for all accepting states $j$.
\end{itemize}
\end{frame}
\begin{frame}[fragile]{Example}
	\begin{code}
zeroStart:: DFA Int
zeroStart = DFA [0,1,2] ['0', '1'] deltazero 0 [1] where
	deltazero:: Symbol -> Int -> Int
	deltazero char st
	| st == 0 && char == '0' = 1
	| st == 0 && char == '1' = 2
	| st == 1 = 1
	| st == 2 = 2
	| otherwise = -1
	\end{code}
